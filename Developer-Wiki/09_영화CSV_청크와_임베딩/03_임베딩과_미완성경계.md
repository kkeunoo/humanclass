```python
def embedding_with_gemini(chunks):
    # 임베딩 모델 가져오기
    api_key = os.getenv('GEMINI_API_KEY')
    client = genai.Client(api_key=api_key)

    actions = []
    for chunk in chunks :
        # 임베딩 하기
        result = client.models.embed_content(
            model='gemini-embedding-2',
            contents=[
                types.Content(
                    parts=[
                        types.Part.from_text(text=chunk['content'])
                    ]
                )
            ],
            # 384, 768은 64의 배수로 딱 떨어지는 수
            config=types.EmbedContentConfig(
                # 디멘션이 달라진다고 해도 실질적인 토큰에는 영향을 주지 않는다
                # 단, 벡터가 많아지기에 저장공간이 커지고 검색이 느려진다
                output_dimensionality=768
            )
        )
        # 임베딩 한 결과 저장
        chunk['embedding'] = result.embeddings[0].values

        # ES용 데이터 형식으로 저장
        actions.append({
            '_index': os.getenv('ES_INDEX'),
            '_id': f'{chunk["movieId"]}-{chunk["chunk_index"]}',
            '_source': chunk
        })
        # gemini가 1분당 제한이 있기 때문에 0.7초로 sleep을 걸어줌
        time.sleep(0.7)

    return actions
```

내 함수는 청크의 `content`를 임베딩 요청에 넣고 응답 벡터를 `chunk['embedding']`에 넣는다. 설정한 차원은 768이며 `_index`, `_id`, `_source`를 갖는 **벌크용 action 목록**을 반환한다. 이것을 반환하는 것과 서버에 벌크 적재하는 것은 다르다. 함수는 청크마다 0.7초를 쉬지만 그 간격이 외부 서비스 제한을 항상 만족한다는 보장은 없다. 환경변수의 **이름**만 코드에 있고 실제 키 값은 Wiki에 옮기지 않는다.

```python
def embedding_with_openai(chunks):
    # 임베딩 모델 가져오기
    api_key = os.getenv('OPENAI_API_KEY')
    client = OpenAI(api_key=api_key)

    actions = []
    last_index = 0
    time_before = time.time()

    contents = [ chunk['content'] for chunk in chunks ]

    for idx_start in range(0, len(contents), 2047) :
        contents_part = contents[idx_start: idx_start + 2047]

        # 임베딩 하기
        # OpenAI의 경우 list를 주면 결과도 list로 나온다
        result = client.embeddings.create(
            model='text-embedding-3-small',
            input=contents_part,
            encoding_format='float', # 기본값: base64 형태(양방향 암호화)
            # dimensions=1
            dimensions=768
        )
```

내 함수는 `content`를 모아 최대 2047개씩 요청하고, 응답 `data`를 `index` 순서로 정렬해 원래 청크에 벡터를 넣는다. 각 청크에 대한 벌크용 action을 만들어 반환한다. 즉 **함수는 완성된 형태로 작성돼 있으나 실제 API 성공, 저장, 검색 결과는 검증되지 않았다.** 원본 주석의 “base64 = 양방향 암호화”는 잘못된 비유다. Base64는 데이터 표현 방식이며 암호화가 아니다.

두 내 함수는 입력 청크 딕셔너리를 제자리에서 바꾸므로 같은 `chunks`를 다시 사용하면 `embedding`이 추가된 상태다. 또한 `_id`가 `movieId-chunk_index`라 같은 영화를 다시 준비하면 같은 ID가 나온다. 실제 서버 쓰기 전에 내용을 확인해야 한다.
